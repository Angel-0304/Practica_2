#Interfaces

 description WAN hacia ISP
 ip address 20.24.23.2 255.255.255.252
 ip nat outside
 no shutdown
exit

interface Ethernet0/1
 description Trunk hacia SW-Usuarios
 no ip address
 no shutdown
exit

interface Ethernet0/1.10
 description VLAN 10 - Usuarios
 encapsulation dot1Q 10
 ip address 10.23.56.1 255.255.255.128
 ip nat inside
exit

#Ruta por defecto hacia el ISP
ip route 0.0.0.0 0.0.0.0 20.24.23.1

#NAT (excluye el trafico de la VPN)
ip access-list extended NAT-LAN
 deny   ip 10.23.56.0 0.0.0.127 10.23.56.128 0.0.0.15
 permit ip 10.23.56.0 0.0.0.127 any
exit
ip nat inside source list NAT-LAN interface Ethernet0/0 overload

# ============================================================
# VPN SITE-TO-SITE IPsec hacia el FortiGate
# ============================================================

#Phase 1 (ISAKMP)
# (des-md5 para coincidir con la licencia de evaluacion del FortiGate)
crypto isakmp policy 10
 encryption des
 hash md5
 authentication pre-share
 group 14
 lifetime 86400
exit
crypto isakmp key <PRE-SHARED-KEY> address 20.24.56.2

#Phase 2 (IPsec)
crypto ipsec transform-set TS-FORTIGATE esp-des esp-md5-hmac
 mode tunnel
exit

#Trafico interesante (selectores de la VPN)
ip access-list extended VPN-TRAFICO
 permit ip 10.23.56.0 0.0.0.127 10.23.56.128 0.0.0.15
exit

#Crypto map
crypto map CMAP-FORTIGATE 10 ipsec-isakmp
 set peer 20.24.56.2
 set transform-set TS-FORTIGATE
 set pfs group14
 set security-association lifetime seconds 43200
 match address VPN-TRAFICO
exit

interface Ethernet0/0
 crypto map CMAP-FORTIGATE
exit

line console 0
 logging synchronous
 exec-timeout 0 0
exit

end
write memory

FG-Server (FortiGate VM64-KVM v6.4.0)
=========================================================================

#Paso 0: acceso de gestion (unico paso por consola)
# (Se uso "set defaultgw disable" para no tomar la ruta por defecto de Cloud0)
config system interface
    edit "port3"
        set mode dhcp
        set defaultgw disable
        set allowaccess ping https ssh http
    next
end

#Hostname (GUI: System > Settings)
config system global
    set hostname "FG-Server"
end

#Interfaces (GUI: Network > Interfaces)
config system interface
    edit "port1"
        set alias "WAN"
        set mode static
        set ip 20.24.56.2 255.255.255.252
        set allowaccess ping
        set role wan
    next
    edit "port2"
        set mode static
        set ip 10.23.56.129 255.255.255.240
        set allowaccess ping
        set role lan
    next
end

#Ruta por defecto hacia el ISP (GUI: Network > Static Routes)
config router static
    edit 1
        set gateway 20.24.56.1
        set device "port1"
    next
end

#Politica de salida con NAT (GUI: Policy & Objects > Firewall Policy)
config firewall policy
    edit 1
        set name "LAN-a-Internet"
        set srcintf "port2"
        set dstintf "port1"
        set srcaddr "all"
        set dstaddr "all"
        set action accept
        set schedule "always"
        set service "ALL"
        set nat enable
    next
end

# ============================================================
# VPN SITE-TO-SITE (GUI: VPN > IPsec Wizard > Site to Site - Cisco)
# ============================================================
config firewall address
    edit "VPN-A-CISCO_local_subnet_1"
        set subnet 10.23.56.128 255.255.255.240
    next
    edit "VPN-A-CISCO_remote_subnet_1"
        set subnet 10.23.56.0 255.255.255.128
    next
end

config firewall addrgrp
    edit "VPN-A-CISCO_local"
        set member "VPN-A-CISCO_local_subnet_1"
        set comment "VPN: VPN-A-CISCO (Created by VPN wizard)"
    next
    edit "VPN-A-CISCO_remote"
        set member "VPN-A-CISCO_remote_subnet_1"
        set comment "VPN: VPN-A-CISCO (Created by VPN wizard)"
    next
end

#Phase 1
#(Propuesta des-md5, limitada por la licencia de evaluacion)
config vpn ipsec phase1-interface
    edit "VPN-A-CISCO"
        set interface "port1"
        set ike-version 1
        set mode main
        set peertype any
        set net-device disable
        set proposal des-md5 des-sha1
        set dhgrp 14 5
        set keylife 86400
        set remote-gw 20.24.23.2
        set psksecret <PRE-SHARED-KEY>
        set comments "VPN: VPN-A-CISCO (Created by VPN wizard)"
    next
end

#Phase 2
config vpn ipsec phase2-interface
    edit "VPN-A-CISCO"
        set phase1name "VPN-A-CISCO"
        set proposal des-md5 des-sha1
        set pfs enable
        set dhgrp 14 5
        set keylifeseconds 43200
        set src-addr-type name
        set dst-addr-type name
        set src-name "VPN-A-CISCO_local"
        set dst-name "VPN-A-CISCO_remote"
    next
end

#Ruta hacia la red remota por el tunel + ruta blackhole
config router static
    edit 2
        set device "VPN-A-CISCO"
        set dstaddr "VPN-A-CISCO_remote"
    next
    edit 3
        set distance 254
        set blackhole enable
        set dstaddr "VPN-A-CISCO_remote"
    next
end

#Politicas de la VPN (sin NAT)
config firewall policy
    edit 2
        set name "vpn_VPN-A-CISCO_local"
        set srcintf "port2"
        set dstintf "VPN-A-CISCO"
        set srcaddr "VPN-A-CISCO_local"
        set dstaddr "VPN-A-CISCO_remote"
        set action accept
        set schedule "always"
        set service "ALL"
    next
    edit 3
        set name "vpn_VPN-A-CISCO_remote"
        set srcintf "VPN-A-CISCO"
        set dstintf "port2"
        set srcaddr "VPN-A-CISCO_remote"
        set dstaddr "VPN-A-CISCO_local"
        set action accept
        set schedule "always"
        set service "ALL"
    next
end
